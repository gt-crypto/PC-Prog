#include <stdio.h>
#include <mpi.h>
int main(int argc,char **argv){
    int r,x,mx,mn,sum,prod;
    MPI_Init(&argc,&argv);
    MPI_Comm_rank(MPI_COMM_WORLD,&r);
    x=r+1;
    MPI_Reduce(&x,&mx,1,MPI_INT,MPI_MAX,0,MPI_COMM_WORLD);
    MPI_Reduce(&x,&mn,1,MPI_INT,MPI_MIN,0,MPI_COMM_WORLD);
    MPI_Reduce(&x,&sum,1,MPI_INT,MPI_SUM,0,MPI_COMM_WORLD);
    MPI_Allreduce(&x,&prod,1,MPI_INT,MPI_PROD,MPI_COMM_WORLD);
    if(r==0)printf("%d %d %d\n",mx,mn,sum);
    printf("%d: %d\n",r,prod);
    MPI_Finalize();
}
