#include <stdio.h>
#include <omp.h>
int prime(int x){
    if(x<2)return 0;
    for(int d=2;d*d<=x;d++)
        if(x%d==0)return 0;
    return 1;
}
int main(){
    int n;
    scanf("%d",&n);
    double t=omp_get_wtime();
    for(int i=2;i<=n;i++)
        if(prime(i))printf("%d ",i);
    printf("\nSerial: %f\n",omp_get_wtime()-t);
    t=omp_get_wtime();
    #pragma omp parallel for
    for(int i=2;i<=n;i++)
        if(prime(i)){
            #pragma omp critical
            printf("%d ",i);
        }
    printf("\nParallel: %f\n",omp_get_wtime()-t);
}
