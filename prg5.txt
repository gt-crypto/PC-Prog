#include <stdio.h>
#include <mpi.h>
int main(int argc,char **argv){
    int rank,x=42;
    MPI_Init(&argc,&argv);
    MPI_Comm_rank(MPI_COMM_WORLD,&rank);
    if(rank==0)
        MPI_Send(&x,1,MPI_INT,1,7,MPI_COMM_WORLD);
    if(rank==1){
        MPI_Recv(&x,1,MPI_INT,0,7,MPI_COMM_WORLD,MPI_STATUS_IGNORE);
        printf("%d\n",x);
    }
    MPI_Finalize();
}
