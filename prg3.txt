#include <stdio.h>
#include <omp.h>
long fib(int n){
    if(n<2)return n;
    long x,y;
    #pragma omp task shared(x)
    x=fib(n-1);
    #pragma omp task shared(y)
    y=fib(n-2);
    #pragma omp taskwait
    return x+y;
}
int main(){
    int n;
    long ans;
    scanf("%d",&n);
    #pragma omp parallel
    #pragma omp single
    ans=fib(n);
    printf("%ld\n",ans);
}
