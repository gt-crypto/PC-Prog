#include <stdio.h>
#include <stdlib.h>
#include <omp.h>

void merge(int a[],int l,int m,int r){
    int t[10000],i=l,j=m+1,k=0;
    while(i<=m&&j<=r)t[k++]=a[i]<a[j]?a[i++]:a[j++];
    while(i<=m)t[k++]=a[i++];
    while(j<=r)t[k++]=a[j++];
    for(i=l,k=0;i<=r;i++)a[i]=t[k++];
}
void sort(int a[],int l,int r){
    if(l>=r)return;
    int m=(l+r)/2;
    sort(a,l,m);sort(a,m+1,r);merge(a,l,m,r);
}
void psort(int a[],int l,int r,int d){
    if(l>=r)return;
    int m=(l+r)/2;
    if(d<2){
        #pragma omp parallel sections
        {
            #pragma omp section
            psort(a,l,m,d+1);
            #pragma omp section
            psort(a,m+1,r,d+1);
        }
    }else{
        sort(a,l,m);sort(a,m+1,r);
    }
    merge(a,l,m,r);
}
int main(){
    int a[10000],b[10000],n;
    scanf("%d",&n);
    for(int i=0;i<n;i++)a[i]=b[i]=rand()%1000;
    double t=omp_get_wtime();
    sort(a,0,n-1);
    printf("Sequential: %f\n",omp_get_wtime()-t);
    t=omp_get_wtime();
    psort(b,0,n-1,0);
    printf("Parallel: %f\n",omp_get_wtime()-t);
}
